import carla
import osmium
import json

class OSMToCourseGenerator:
    def __init__(self, osm_file_path: str, output_dir: str):
        self.osm_file = osm_file_path
        self.output_dir = output_dir
        
    def convert_to_opendrive(self) -> str:
        """Convert OSM to OpenDRIVE using CARLA's stable API."""
        settings = carla.Osm2OdrSettings()
        settings.use_offsets = True
        settings.generate_traffic_lights = True
        settings.generate_traffic_signs = True
        settings.set_osm_way_types = ["motorway", "trunk", "primary", "secondary", "tertiary", "residential"]
        settings.default_lane_width = 3.5  # meters
        
        # Read OSM file
        with open(self.osm_file, 'r', encoding='utf-8') as f:
            osm_data = f.read()
        
        # Convert
        xodr_data = carla.Osm2Odr.convert(osm_data, settings)
        
        # Save
        xodr_path = os.path.join(self.output_dir, "course.xodr")
        with open(xodr_path, 'w') as f:
            f.write(xodr_data)
        
        return xodr_path
    
    def validate_course(self, xodr_path: str) -> dict:

4.3.2 Course Validation Service (Server-Side)
Objective: Validate submitted courses for approval.

java
@Service
public class CourseValidationService {
    
    private static final double MIN_LENGTH_KM = 1.0;
    private static final double MAX_LENGTH_KM = 15.0;
    private static final double MIN_AVG_SPEED_KPH = 40.0;
    private static final int MAX_COLLISIONS = 3;
    
    @Autowired
    private CourseRepository courseRepository;
    
    @Autowired
    private PlayerService playerService;
    
    public ValidationResult validateCourse(CourseSubmission submission) {
        // 1. Parse telemetry
        List<GeoPoint> path = parseTelemetry(submission.getTelemetry());
        
        // 2. Check length
        double lengthKm = calculatePathLength(path);
        if (lengthKm < MIN_LENGTH_KM || lengthKm > MAX_LENGTH_KM) {
            return ValidationResult.failed("Course length outside acceptable range: " + lengthKm + "km");
        }
        
        // 3. Check connectivity
        boolean isConnected = isPathConnected(path);
        if (!isConnected) {
            return ValidationResult.failed("Course path is not fully connected");
        }
        
        // 4. Check loop detection
        boolean isLoop = detectLoop(path);
        if (!isLoop) {
            return ValidationResult.failed("Course must be a loop (start and end at same point)");
        }
        
        // 5. Check average speed
        double avgSpeed = calculateAverageSpeed(submission.getTelemetry());
        if (avgSpeed < MIN_AVG_SPEED_KPH) {
            return ValidationResult.failed("Average speed too low: " + avgSpeed + " km/h");
        }
        
        // 6. Check collision count
        int collisionCount = submission.getCollisionCount();
        if (collisionCount > MAX_COLLISIONS) {
            return ValidationResult.failed("Too many collisions: " + collisionCount);
        }
        
        // 7. Check for duplicates (within 50m)
        boolean isDuplicate = courseRepository.existsNearPath(path, 50.0);
        if (isDuplicate) {
            return ValidationResult.failed("Similar course already exists nearby");
        }
        
        // 8. Generate tags
        List<String> tags = generateTags(path, submission.getTelemetry());
        
        // 9. Calculate difficulty score
        double difficulty = calculateDifficulty(path, submission.getTelemetry());
        
        // 10. Create approved course
        Course course = new Course();
        course.setCreatorId(submission.getPlayerId());
        course.setPath(path);
        course.setLengthKm(lengthKm);
        course.setStartPoint(path.get(0));
        course.setEndPoint(path.get(path.size() - 1));
        course.setZipCode(submission.getZipCode());
        course.setCity(submission.getCity());
        course.setApproved(true);
        course.setTags(tags);
        course.setDifficulty(difficulty);
        course.setRaceCount(0);
        course.setRatingAvg(0.0);
        course.setCreatedAt(Instant.now());
        
        courseRepository.save(course);
        
        // 11. Verify player
        Player player = playerService.getPlayer(submission.getPlayerId());
        if (!player.isVerified()) {
            player.setVerified(true);
            playerService.updatePlayer(player);
        }
        
        return ValidationResult.success(course);
    }
    
    private List<String> generateTags(List<GeoPoint> path, TelemetryData telemetry) {
        List<String> tags = new ArrayList<>();
        
        // Urban detection
        double avgBuildingDensity = getAverageBuildingDensity(path);
        if (avgBuildingDensity > 0.7) {
            tags.add("urban");
        }
        
        // Twisty detection
        double twistyScore = calculateTwistyScore(path);
        if (twistyScore > 0.6) {
            tags.add("twisty");
        } else {
            tags.add("highway");
        }
        
        // Elevation detection
        double elevationGain = calculateElevationGain(path);
        if (elevationGain > 50.0) {
            tags.add("hilly");
        }
        
        // Length tags
        double lengthKm = calculatePathLength(path);
        if (lengthKm < 3.0) {
            tags.add("short");
        } else if (lengthKm < 6.0) {
            tags.add("medium");
        } else {
            tags.add("long");
        }
        
        // Difficulty
        double difficulty = calculateDifficulty(path, telemetry);
        if (difficulty < 3.0) {
            tags.add("easy");
        } else if (difficulty < 6.0) {
            tags.add("medium");
        } else {
            tags.add("hard");
        }
        
        return tags;
    }
}

@Service
@Transactional
public class ChallengeService {
    
    @Autowired
    private ChallengeRepository challengeRepository;
    
    @Autowired
    private PlayerService playerService;
    
    @Autowired
    private RaceSessionService raceSessionService;
    
    @Autowired
    private NotificationService notificationService;
    
    @Autowired
    private WagerService wagerService;
    
    private static final Duration CHALLENGE_EXPIRY = Duration.ofHours(24);
    private static final int MAX_PENDING_CHALLENGES = 3;
    
    public Challenge createChallenge(CreateChallengeRequest request) {
        // 1. Validate players
        Player challenger = playerService.getPlayer(request.getChallengerId());
        Player defender = playerService.getPlayer(request.getDefenderId());
        
        if (!challenger.isVerified() || !defender.isVerified()) {
            throw new IllegalStateException("Both players must be verified");
        }
        
        // 2. Validate course
        Course course = courseRepository.findById(request.getCourseId())
            .orElseThrow(() -> new IllegalArgumentException("Course not found"));
        
        if (!course.isApproved()) {
            throw new IllegalStateException("Course not approved for racing");
        }
        
        // 3. Validate wager (pink slip)
        WagerValidation wagerValidation = wagerService.validateWager(
            request.getChallengerId(), 
            request.getWagerType(), 
            request.getWagerValue()
        );
        
        if (!wagerValidation.isValid()) {
            throw new IllegalArgumentException(wagerValidation.getError());
        }
        
        // 4. Check defender's pending challenges
        long pendingCount = challengeRepository.countByDefenderIdAndStatus(
            request.getDefenderId(), 
            ChallengeStatus.PENDING
        );
        if (pendingCount >= MAX_PENDING_CHALLENGES) {
            throw new IllegalStateException("Defender has too many pending challenges");
        }
        
        // 5. Create challenge
        Challenge challenge = new Challenge();
        challenge.setId(UUID.randomUUID());
        challenge.setChallengerId(request.getChallengerId());
        challenge.setDefenderId(request.getDefenderId());
        challenge.setCourseId(request.getCourseId());
        challenge.setWagerType(request.getWagerType());
        challenge.setWagerValue(request.getWagerValue());
        challenge.setStatus(ChallengeStatus.PENDING);
        challenge.setCreatedAt(Instant.now());
        challenge.setExpiresAt(Instant.now().plus(CHALLENGE_EXPIRY));
        
        Challenge saved = challengeRepository.save(challenge);
        
        // 6. Send notification to defender
        Notification notification = Notification.builder()
            .type(NotificationType.CHALLENGE_RECEIVED)
            .playerId(request.getDefenderId())
            .data(Map.of(
                "challenge_id", saved.getId().toString(),
                "challenger_name", challenger.getUsername(),
                "course_name", course.getName(),
                "wager", request.getWagerValue(),
                "expires_in", CHALLENGE_EXPIRY.toString()
            ))
            .build();
        notificationService.sendNotification(notification);
        
        return saved;
    }
    
    public RaceSession acceptChallenge(UUID challengeId, UUID defenderId) {
        Challenge challenge = getChallenge(challengeId);
        
        // 1. Validate challenge state
        if (challenge.getStatus() != ChallengeStatus.PENDING) {
            throw new IllegalStateException("Challenge is not pending");
        }
        
        if (!challenge.getDefenderId().equals(defenderId)) {
            throw new IllegalArgumentException("Defender ID mismatch");
        }
        
        if (challenge.getExpiresAt().isBefore(Instant.now())) {
            challenge.setStatus(ChallengeStatus.EXPIRED);
            challengeRepository.save(challenge);
            throw new IllegalStateException("Challenge has expired");
        }
        
        // 2. Validate both players are online
        boolean challengerOnline = playerService.isOnline(challenge.getChallengerId());
        boolean defenderOnline = playerService.isOnline(defenderId);
        
        if (!challengerOnline || !defenderOnline) {
            throw new IllegalStateException("Both players must be online to accept challenge");
        }
        
        // 3. Create race session
        RaceSession session = raceSessionService.createSession(
            challenge.getChallengerId(),
            defenderId,
            challenge.getCourseId()
        );
        
        // 4. Update challenge
        challenge.setStatus(ChallengeStatus.ACCEPTED);
        challenge.setSessionId(session.getId());
        challengeRepository.save(challenge);
        
        // 5. Notify both players
        notificationService.broadcastToPlayers(
            List.of(challenge.getChallengerId(), defenderId),
            NotificationType.RACE_STARTING,
            Map.of(
                "session_id", session.getId(),
                "course_id", challenge.getCourseId()
            )
        );
        
        return session;
    }
    
    public RaceResult resolveChallenge(UUID challengeId, RaceResult raceResult) {
        Challenge challenge = getChallenge(challengeId);
        
        if (challenge.getStatus() != ChallengeStatus.IN_PROGRESS) {
            throw new IllegalStateException("Challenge is not in progress");
        }
        
        // 1. Determine winner
        UUID winnerId = raceResult.getWinnerId();
        UUID loserId = raceResult.getLoserId();
        
        // 2. Update player stats
        Player winner = playerService.getPlayer(winnerId);
        Player loser = playerService.getPlayer(loserId);
        
        winner.setWins(winner.getWins() + 1);
        winner.setRating(winner.getRating() + 10);
        loser.setLosses(loser.getLosses() + 1);
        loser.setRating(Math.max(0, loser.getRating() - 10));
        
        playerService.updatePlayer(winner);
        playerService.updatePlayer(loser);
        
        // 3. Handle pink slip wager
        if ("pink_slip".equals(challenge.getWagerType())) {
            String wagerCarId = challenge.getWagerValue().get("car_id").asText();
            wagerService.transferCar(loserId, winnerId, wagerCarId);
        }
        
        // 4. Complete challenge
        challenge.setStatus(ChallengeStatus.COMPLETED);
        challenge.setCompletedAt(Instant.now());
        challenge.setRaceResult(raceResult);
        challengeRepository.save(challenge);
        
        // 5. Update course stats
        Course course = courseRepository.findById(challenge.getCourseId()).orElseThrow();
        course.setRaceCount(course.getRaceCount() + 1);
        double newRating = (course.getRatingAvg() * (course.getRaceCount() - 1) 
            + raceResult.getRaceQuality()) / course.getRaceCount();
        course.setRatingAvg(newRating);
        courseRepository.save(course);
        
        return raceResult;
    }
}

8. Day 1 Critical Bugs & Mitigation
8.1 Identified Critical Issues
Bug #1: CARLA OSM Import Memory Overflow
Description: When importing large OSM files (>50MB) through carla.Osm2Odr, the Python process crashes with memory overflow.

Root Cause: The OSM parser attempts to load the entire file into memory before processing.

Mitigation Strategy:

python
import osmium
import carla

class OSMStreamingParser(osmium.SimpleHandler):
    """Streaming OSM parser to avoid memory overflow."""
    
    def __init__(self):
        super().__init__()
        self.ways = []
        self.nodes = {}
    
    def way(self, w):
        # Process each way as it's read
        if self.is_road(w):
            self.ways.append(w)
        # Flush if memory exceeds threshold
        if len(self.ways) > 10000:
            self.flush_and_convert()
    
    def flush_and_convert(self):
        """Convert accumulated ways to OpenDRIVE and flush."""
        # Convert batch
        xodr_data = carla.Osm2Odr.convert_batch(self.ways, self.nodes)
        # Write to temporary file
        self.write_to_temp(xodr_data)
        # Clear memory
        self.ways = []
        self.nodes = {}

# Use streaming parser
parser = OSMStreamingParser()
parser.apply_file("large_city.osm", locations=True)

Bug #3: OpenDRIVE Parser Lacks Bridge/Overpass Support
Description: CARLA's OpenDRIVE parser cannot load maps containing bridges or overpasses.

Root Cause: OpenDRIVE's complex road geometry is not fully supported.

Mitigation Strategy:

Pre-filter OSM data to remove overpass/bridge segments

python
def filter_osd_for_opendrive(osm_data):
    """Remove bridge/overpass tags from OSM data."""
    # Remove ways with bridge=yes or tunnel=yes
    filtered = osm_data.replace('bridge="yes"', 'bridge="no"')
    filtered = filtered.replace('tunnel="yes"', 'tunnel="no"')
    return filtered

void FixSplineNormals(USplineComponent* Spline)
{
    int NumPoints = Spline->GetNumberOfSplinePoints();
    for (int i = 0; i < NumPoints; i++)
    {
        // Check if normal points downward
        FVector Up = Spline->GetUpVectorAtSplinePoint(i, ESplineCoordinateSpace::World);
        if (Up.Z < 0)
        {
            // Flip the normal
            Spline->SetUpVectorAtSplinePoint(i, -Up, ESplineCoordinateSpace::World);
        }
    }
    Spline->UpdateSpline();
}
